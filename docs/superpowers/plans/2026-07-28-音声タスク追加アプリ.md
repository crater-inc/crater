# 音声タスク追加アプリ Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** ブラウザで録音→文字起こしした内容を、話し終わったら即座に指定のGoogleスプレッドシートの2行目（ヘッダー直下）に新規タスク行として追加するWebアプリを作る。

**Architecture:** 単一HTML（Web Speech APIで文字起こし）が、対象スプレッドシートに紐付けたGoogle Apps Script製Web App（`doPost`）へ`fetch`でPOSTし、GAS側がシートに`insertRowBefore(2)`で行を挿入する。GitHub Pages（`view.crater.co.jp`）でホスティングし、Firebase等の追加インフラは使わない。

**Tech Stack:** HTML/CSS/JavaScript（Web Speech API）、Google Apps Script（Web App / `SpreadsheetApp`）、GitHub Pages。

## Global Constraints

- 対象スプレッドシートID：`1nZ5A991aTfNPNZdSDq2XWcUj55lMT7qhGkQCfD7P0s4`、対象シート：`gid=0`
- 挿入位置：常にヘッダー直下＝2行目。既存データ行は一切削除しない（下にシフトする）
- 挿入する値：B列=`0`固定、C列=`0`固定、D列=`01_TODO`固定、E列=話した内容そのまま、F列=空欄
- 送信フロー：録音停止と同時に確認画面なしで即送信（プレビュー・修正画面は作らない）
- アクセス制御：GAS Web Appは「実行ユーザー：自分」「アクセスできるユーザー：自分のみ」でデプロイする
- 録音ボタンの色：Googleスプレッドシートのブランドグリーン（`#0f9d58`系）
- 見出しフォントは`urw-din`（Adobe Typekit kit `qwe2tdt`）、既存の音声メモアプリ（`音声メモアプリ/index.html`）と同系統のダークUIに揃える
- Firebase・Cloudflare Workers等の追加インフラは使わない（YAGNI）
- 全UIテキストは日本語
- ファイル・フォルダ名は日本語（`音声タスク追加アプリ/`）

- [ ] **Step 1: Apps Scriptのコードをローカルに書く**

`音声タスク追加アプリ/gas/コード.gs` を作成:

```javascript
// 音声タスク追加アプリ：受け取ったテキストを対象シートの2行目に新規タスクとして挿入する
const SPREADSHEET_ID = '1nZ5A991aTfNPNZdSDq2XWcUj55lMT7qhGkQCfD7P0s4';
const TARGET_SHEET_GID = 0;

function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents);
    const text = (body.text || '').trim();
    if (!text) {
      return jsonOutput_({ ok: false, error: 'empty text' });
    }
    const sheet = getTargetSheet_();
    sheet.insertRowBefore(2);
    sheet.getRange(2, 2, 1, 4).setValues([[0, 0, '01_TODO', text]]);
    return jsonOutput_({ ok: true });
  } catch (err) {
    return jsonOutput_({ ok: false, error: String(err) });
  }
}

function getTargetSheet_() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheets = ss.getSheets();
  for (let i = 0; i < sheets.length; i++) {
    if (sheets[i].getSheetId() === TARGET_SHEET_GID) return sheets[i];
  }
  throw new Error('target sheet not found (gid=' + TARGET_SHEET_GID + ')');
}

function jsonOutput_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
```

const GAS_URL = '__GAS_WEBAPP_URL__'; // Task1でデプロイしたURLに置き換え済みであること

let recognition = null;
let finalText = '';
let interimText = '';
let recording = false;
let sending = false;
let lastFailedText = '';

function speechSupported() {
  return 'webkitSpeechRecognition' in window || 'SpeechRecognition' in window;
}

function showScreen(id) {
  document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
  document.getElementById(id).classList.add('active');
}

function goRecord() {
  if (!speechSupported()) {
    alert('このブラウザは音声認識に対応していません。Google Chrome で開いてください。');
    return;
  }
  showScreen('record');
  startRecording();
}

function startRecording() {
  finalText = '';
  interimText = '';
  document.getElementById('transcript').innerHTML = '<span class="placeholder">ここに話した内容がテキストで出ます</span>';
  document.getElementById('recStatus').innerHTML = '<span class="pulse"></span>録音中… 話してください';
  document.getElementById('recActions').style.display = 'flex';
  document.getElementById('stopBtn').disabled = false;

  const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
  recognition = new SR();
  recognition.lang = 'ja-JP';
  recognition.continuous = true;
  recognition.interimResults = true;

  recognition.onresult = (e) => {
    let interim = '';
    for (let i = e.resultIndex; i < e.results.length; i++) {
      const t = e.results[i][0].transcript;
      if (e.results[i].isFinal) finalText += t;
      else interim += t;
    }
    interimText = interim;
    renderTranscript(interim);
  };
  recognition.onerror = (e) => {
    if (e.error === 'not-allowed' || e.error === 'service-not-allowed') {
      document.getElementById('recStatus').innerHTML = '⚠️ マイクの使用が許可されていません';
    }
  };
  recognition.onend = () => { if (recording) { try { recognition.start(); } catch(e){} } };

  recording = true;
  try { recognition.start(); } catch(e){}
}

function renderTranscript(interim) {
  const el = document.getElementById('transcript');
  if (!finalText && !interim) {
    el.innerHTML = '<span class="placeholder">ここに話した内容がテキストで出ます</span>';
    return;
  }
  el.innerHTML = escapeHtml(finalText) + '<span class="interim">' + escapeHtml(interim) + '</span>';
  el.scrollTop = el.scrollHeight;
}

function stopRecognition() {
  recording = false;
  if (recognition) { try { recognition.stop(); } catch(e){} recognition = null; }
}

async function stopAndSend() {
  stopRecognition();
  const el = document.getElementById('transcript');
  let domText = '';
  if (el && !el.querySelector('.placeholder')) domText = (el.innerText || el.textContent || '').trim();
  const text = (domText || (finalText + interimText)).trim();
  if (!text) { toast('内容が空でした', true); showScreen('home'); return; }
  await send(text);
}

async function send(text) {
  if (sending) return;
  sending = true;
  document.getElementById('stopBtn').disabled = true;
  toast('追加中…');
  try {
    const res = await fetch(GAS_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'text/plain;charset=utf-8' },
      body: JSON.stringify({ text })
    });
    const data = await res.json();
    if (data.ok) {
      toast('追加しました');
      setTimeout(() => showScreen('home'), 1200);
    } else {
      toast('追加に失敗しました: ' + (data.error || '不明なエラー'), true);
      showRetry(text);
    }
  } catch (err) {
    toast('通信に失敗しました', true);
    showRetry(text);
  }
  sending = false;
}

function showRetry(text) {
  lastFailedText = text;
  document.getElementById('recActions').style.display = 'none';
  document.getElementById('recStatus').innerHTML = '⚠️ 送信できませんでした';
  document.getElementById('transcript').innerHTML =

    '<div style="margin-top:16px; display:flex; gap:12px;">' +

    '<button class="btn-cancel" style="width:auto; padding:10px 20px; border-radius:12px;" onclick="cancelRecord()">ホームに戻る</button>' +
    '</div>';
}

function retrySend() { send(lastFailedText); }

function cancelRecord() {
  stopRecognition();
  showScreen('home');
}

function escapeHtml(s){ return (s||'').replace(/[&<>]/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;'}[c])); }
function toast(msg, isError){
  const t = document.getElementById('toast');
  t.textContent = msg;
  t.classList.toggle('error', !!isError);
  t.classList.add('show');
  clearTimeout(t._timer);
  t._timer = setTimeout(() => t.classList.remove('show'), 2500);
}
